cmake_minimum_required(VERSION 3.20)
project(screenplay_tatham LANGUAGES C)
if(NOT PUZZLES_SOURCE)
  message(FATAL_ERROR "Set PUZZLES_SOURCE to the prepared, pinned source tree")
endif()
set(CMAKE_C_STANDARD 99)
string(REPLACE "-DNDEBUG" "" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
include_directories(${PUZZLES_SOURCE} ${CMAKE_CURRENT_BINARY_DIR})
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/generated-games.h "GAME(net)\nGAME(untangle)\nGAME(fifteen)\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/version.h "#define VER \"ScreenPlay pinned 790f585 adapter 0.1\"\n")
set(core_names combi divvy draw-poly dsf findloop grid latin laydomino loopgen
    malloc matching midend misc penrose penrose-legacy ps random sort tdq tree234 version)
set(core_sources)
foreach(name IN LISTS core_names)
  list(APPEND core_sources ${PUZZLES_SOURCE}/${name}.c)
endforeach()
set(game_sources ${PUZZLES_SOURCE}/net.c ${PUZZLES_SOURCE}/untangle.c
                 ${PUZZLES_SOURCE}/fifteen.c ${PUZZLES_SOURCE}/list.c)
add_library(puzzle_core STATIC ${core_sources} ${game_sources})
target_compile_definitions(puzzle_core PUBLIC COMBINED)
add_executable(screenplay-engine engine.c ${PUZZLES_SOURCE}/nullfe.c)
target_link_libraries(screenplay-engine PRIVATE puzzle_core)
if(APPLE)
  enable_language(OBJC)
  add_executable(screenplay-puzzles MACOSX_BUNDLE ${PUZZLES_SOURCE}/osx.m ${PUZZLES_SOURCE}/drawing.c)
  set_target_properties(screenplay-puzzles PROPERTIES OUTPUT_NAME ScreenPlayPuzzles
    MACOSX_BUNDLE_GUI_IDENTIFIER org.screenplay.puzzles
    MACOSX_BUNDLE_BUNDLE_NAME ScreenPlayPuzzles)
  target_compile_options(screenplay-puzzles PRIVATE -Wno-deprecated-declarations)
  target_link_libraries(screenplay-puzzles PRIVATE puzzle_core "-framework Cocoa")
else()
  target_link_libraries(screenplay-engine PRIVATE m)
endif()
